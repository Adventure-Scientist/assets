form <- low_class ~ age + lwt + smoke + ptl

p_glm <- p_lda <- p_qda <- p_lasso <- rep(NA_real_, nrow(birth))

X <- model.matrix(~ age + lwt + smoke + ptl,
                  data = birth)[, -1, drop = FALSE]

y <- birth$low

for (k in seq_len(K)) {

  tr <- which(fold != k)
  te <- which(fold == k)

  # Logistic regression
  mod_glm <- glm(
    form,
    data = birth[tr, , drop = FALSE],
    family = binomial()
  )

  p_glm[te] <- as.numeric(
    predict(mod_glm, newdata = birth[te, , drop = FALSE],
            type = "response")
  )

  # Linear discriminant analysis
  mod_lda <- MASS::lda(
    form,
    data = birth[tr, , drop = FALSE]
  )

  p_lda[te] <- predict(
    mod_lda,
    newdata = birth[te, , drop = FALSE]
  )$posterior[, "Yes"]

  # Quadratic discriminant analysis
  mod_qda <- MASS::qda(
    form,
    data = birth[tr, , drop = FALSE]
  )

  p_qda[te] <- predict(
    mod_qda,
    newdata = birth[te, , drop = FALSE]
  )$posterior[, "Yes"]

  # LASSO logistic regression
  set.seed(2026 + k)

  mod_lasso <- glmnet::cv.glmnet(
    X[tr, , drop = FALSE],
    y[tr],
    family = "binomial",
    alpha = 1,
    nfolds = 5,
    type.measure = "deviance",
    standardize = TRUE
  )

  p_lasso[te] <- as.numeric(
    predict(
      mod_lasso,
      newx = X[te, , drop = FALSE],
      s = "lambda.min",
      type = "response"
    )
  )
}

stopifnot(

score <- function(p, y, threshold = 0.5) {
  stopifnot(
    length(p) == length(y),
    all(is.finite(p)),
    all(p >= 0 & p <= 1),
    all(y %in% c(0, 1))
  )

  eps <- 1e-12
  q <- pmin(pmax(p, eps), 1 - eps)
  pred <- as.integer(p >= threshold)

  c(
    AUC = unname(Hmisc::somers2(p, y)["C"]),
    Brier = mean((p - y)^2),
    LogLoss = -mean(y * log(q) + (1 - y) * log1p(-q)),
    Accuracy = mean(pred == y),
    Sensitivity = mean(pred[y == 1] == 1),
    Specificity = mean(pred[y == 0] == 0)
  )
}

comparison <- rbind(
  Logistic = score(p_glm, y),
  LDA      = score(p_lda, y),
  QDA      = score(p_qda, y),
  LASSO    = score(p_lasso, y)
)
